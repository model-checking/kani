36/// The models below are used by the compiler to generate nondeterministic
37/// `Box<T>`/`Rc<T>`/`Arc<T>` values for automatic harnesses when `T` does not implement
38/// `Arbitrary` in source code but the compiler can derive it: the `kani::any::<T>()` calls in
39/// their bodies are then replaced with the compiler-synthesized implementation. (For `T`s that
40/// do implement `Arbitrary`, the `Arbitrary` implementations above are resolved directly
41/// instead.)
42/// These models are *optional*: they require `alloc` and thus have no `core::kani`
43/// counterpart, c.f. `KaniModel::is_optional`.
44#[kanitool::fn_marker = "AnyBoxModel"]
45#[inline(never)]
46#[doc(hidden)]
47pub fn any_box<T: Arbitrary>() -> Box<T> {
48    Box::new(crate::any())
49}

53#[doc(hidden)]
54pub fn any_rc<T: Arbitrary>() -> std::rc::Rc<T> {

65impl Arbitrary for std::time::Duration {
66    fn any() -> Self {
67        const NANOS_PER_SEC: u32 = 1_000_000_000;
68        let nanos = u32::any();
69        crate::assume(nanos < NANOS_PER_SEC);
70        std::time::Duration::new(u64::any(), nanos)
71    }
72}

74/// Generate a slice of *unbounded* nondeterministic length: a fresh allocation of
75/// nondeterministic size whose contents are nondeterministic, with element validity
76/// established by `slice_validity_assume` (a compiler hook that emits a quantified
77/// assumption constraining each element's raw bits to the element type's layout niche;
78/// a no-op for element types whose every bit pattern is valid, e.g. integers).
79///
80/// This model is used by the compiler to generate nondeterministic `&[T]` arguments for
81/// automatic harnesses (`kani autoharness`) when the element type qualifies; verification
82/// results hold for ALL slice lengths (functions that iterate over the slice surface any
83/// insufficient loop bound as an unwinding-assertion failure rather than passing silently).
84///
85/// This model is *optional*: it requires `alloc` and thus has no `core::kani` counterpart,
86/// c.f. `KaniModel::is_optional`.
87#[kanitool::fn_marker = "AnySliceRefUnboundedModel"]
88#[inline(never)]
89#[doc(hidden)]
90pub fn any_slice_ref_unbounded<T>() -> &'static [T] {
91    let len: usize = crate::any();
92    let elem = std::mem::size_of::<T>();
93    if elem == 0 {
94        // ZST slices: no storage needed, any length is fine.
95        return unsafe { std::slice::from_raw_parts(std::ptr::NonNull::dangling().as_ptr(), len) };
96    }
97    crate::assume(len <= (isize::MAX as usize) / elem);
98    let layout = std::alloc::Layout::array::<T>(len.max(1)).unwrap();
99    let ptr = unsafe { std::alloc::alloc(layout) };
100    crate::assume(!ptr.is_null());
101    slice_validity_assume::<T>(ptr, len);
102    unsafe { std::slice::from_raw_parts(ptr as *const T, len) }
103}

105/// Generate a mutable slice of *unbounded* nondeterministic length: as
106/// `any_slice_ref_unbounded`, but returning `&mut [T]`. Each call produces a fresh (leaked)
107/// allocation, so the returned slice is exclusive by construction; writes through it are
108/// unconstrained by other generated values.
109///
110/// This model is *optional*: it requires `alloc`, c.f. `KaniModel::is_optional`.
111#[kanitool::fn_marker = "AnySliceMutUnboundedModel"]
112#[inline(never)]
113#[doc(hidden)]
114pub fn any_slice_mut_unbounded<T>() -> &'static mut [T] {
115    let len: usize = crate::any();
116    let elem = std::mem::size_of::<T>();
117    if elem == 0 {
118        return unsafe {
119            std::slice::from_raw_parts_mut(std::ptr::NonNull::dangling().as_ptr(), len)
120        };
121    }
122    crate::assume(len <= (isize::MAX as usize) / elem);
123    let layout = std::alloc::Layout::array::<T>(len.max(1)).unwrap();
124    let ptr = unsafe { std::alloc::alloc(layout) };
125    crate::assume(!ptr.is_null());
126    slice_validity_assume::<T>(ptr, len);
127    unsafe { std::slice::from_raw_parts_mut(ptr as *mut T, len) }
128}

130/// Generate a `Vec` of *unbounded* nondeterministic length: a fresh allocation of
131/// nondeterministic size whose contents are nondeterministic, with element validity
132/// established by `slice_validity_assume` (c.f. `any_slice_ref_unbounded`), handed to
133/// `Vec::from_raw_parts` with `capacity` equal to the allocated element count (`len.max(1)`
134/// for non-ZSTs, since the allocation uses `Layout::array::<T>(len.max(1))` to avoid a
135/// zero-sized allocation; `usize::MAX` for ZSTs). The allocation thus came from the global
136/// allocator with exactly the layout `Vec`'s safety contract requires for `capacity`, so `Vec`
137/// frees it correctly on drop.
138///
139/// This model is used by the compiler to generate nondeterministic `Vec<T>` arguments for
140/// automatic harnesses (`kani autoharness`) when the element type qualifies; verification
141/// results hold for ALL lengths. Optional: requires `alloc`.
142#[kanitool::fn_marker = "AnyVecUnboundedModel"]
143#[inline(never)]
144#[doc(hidden)]
145pub fn any_vec_unbounded<T>() -> Vec<T> {
146    let len: usize = crate::any();
147    let elem = std::mem::size_of::<T>();
148    if elem == 0 {
149        // For ZSTs, Vec never allocates and uses a dangling pointer; constructing from a
150        // dangling pointer with any len is the documented pattern (and loop-free, which
151        // matters: generation code must not itself be bounded by unwinding).
152        return unsafe {
153            Vec::from_raw_parts(std::ptr::NonNull::dangling().as_ptr(), len, usize::MAX)
154        };
155    }
156    crate::assume(len <= (isize::MAX as usize) / elem);
157    let layout = std::alloc::Layout::array::<T>(len.max(1)).unwrap();
158    let ptr = unsafe { std::alloc::alloc(layout) };
159    crate::assume(!ptr.is_null());
160    slice_validity_assume::<T>(ptr, len);
161    unsafe { Vec::from_raw_parts(ptr as *mut T, len, len.max(1)) }
162}

164/// Compiler hook (c.f. `KaniHook::SliceValidityAssume`): assume that every element of the
165/// `len`-element `T`-array at `ptr` has raw bits within `T`'s layout niche. Lowered directly
166/// to a quantified goto assumption; a no-op when `T` has no niche. The default body is
167/// unreachable: calls are always intercepted during code generation.
168#[kanitool::fn_marker = "SliceValidityAssumeHook"]
169#[inline(never)]
170#[doc(hidden)]
171pub fn slice_validity_assume<T>(_ptr: *const u8, _len: usize) {
172    #[cfg(not(kani))]
173    unreachable!("kani::slice_validity_assume is a verification-only hook");
174}

176/// Nondeterministic functions for instantiating `Fn`/`FnMut`/`FnOnce`-bounded type
177/// parameters of automatic harnesses: the parameter is instantiated with the *function
178/// item type* of the matching-arity model below (function items implement all three `Fn`
179/// traits and are zero-sized, so generating the value is trivial). Each call returns a
180/// fresh nondeterministic value, which over-approximates the behavior of every real
181/// closure with that signature (including stateful `FnMut` closures); verifying the
182/// harness against this instantiation therefore covers the function-under-test's own code
183/// for any closure behavior.
184///
185/// These models are *optional* (c.f. `KaniModel::is_optional`).
186#[kanitool::fn_marker = "NondetFn0Model"]
187#[inline(never)]
188#[doc(hidden)]
189pub fn nondet_fn0<R: Arbitrary>() -> R {
190    crate::any()
191}
